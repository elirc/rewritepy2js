import { Router, Response, NextFunction } from 'express';
import { ArticleRepository } from '../repositories/ArticleRepository';
import { CommentRepository } from '../repositories/CommentRepository';
import { authenticate, optionalAuthenticate, AuthenticatedRequest } from '../middleware/auth';
import { validate } from '../middleware/validate';
import { NewArticleSchema, UpdateArticleSchema, NewCommentSchema } from '../schemas';

const router = Router();

// 1. List Articles
router.get('/articles', optionalAuthenticate, async (req: AuthenticatedRequest, res: Response, next: NextFunction): Promise<void> => {
  try {
    const currentUserId = req.user?.id;
    const result = await ArticleRepository.list(req.query, currentUserId);
    res.status(200).json(result);
  } catch (error) {
    next(error);
  }
});

// 2. Feed Articles
router.get('/articles/feed', authenticate, async (req: AuthenticatedRequest, res: Response, next: NextFunction): Promise<void> => {
  try {
    const currentUserId = req.user!.id;
    const result = await ArticleRepository.feed(req.query, currentUserId);
    res.status(200).json(result);
  } catch (error) {
    next(error);
  }
});

// 3. Create Article
router.post('/articles', authenticate, validate(NewArticleSchema), async (req: AuthenticatedRequest, res: Response, next: NextFunction): Promise<void> => {
  try {
    const currentUserId = req.user!.id;
    const article = await ArticleRepository.create(req.body.article, currentUserId);
    res.status(201).json({ article });
  } catch (error) {
    next(error);
  }
});

// 4. Get Article
router.get('/articles/:slug', optionalAuthenticate, async (req: AuthenticatedRequest, res: Response, next: NextFunction): Promise<void> => {
  try {
    const currentUserId = req.user?.id;
    const article = await ArticleRepository.getBySlug(req.params.slug, currentUserId);
    if (!article) {
      res.status(404).json({ errors: { body: ['Article not found'] } });
      return;
    }
    res.status(200).json({ article });
  } catch (error) {
    next(error);
  }
});

// 5. Update Article
router.put('/articles/:slug', authenticate, validate(UpdateArticleSchema), async (req: AuthenticatedRequest, res: Response, next: NextFunction): Promise<void> => {
  try {
    const currentUserId = req.user!.id;
    const article = await ArticleRepository.update(req.params.slug, req.body.article, currentUserId);
    if (!article) {
      res.status(404).json({ errors: { body: ['Article not found'] } });
      return;
    }
    res.status(200).json({ article });
  } catch (error: any) {
    if (error.message === 'Forbidden') {
      res.status(403).json({ errors: { body: ['You are not authorized to update this article'] } });
    } else {
      next(error);
    }
  }
});

// 6. Delete Article
router.delete('/articles/:slug', authenticate, async (req: AuthenticatedRequest, res: Response, next: NextFunction): Promise<void> => {
  try {
    const currentUserId = req.user!.id;
    const success = await ArticleRepository.delete(req.params.slug, currentUserId);
    if (!success) {
      res.status(404).json({ errors: { body: ['Article not found'] } });
      return;
    }
    res.status(204).end();
  } catch (error: any) {
    if (error.message === 'Forbidden') {
      res.status(403).json({ errors: { body: ['You are not authorized to delete this article'] } });
    } else {
      next(error);
    }
  }
});

// 7. Favorite Article
router.post('/articles/:slug/favorite', authenticate, async (req: AuthenticatedRequest, res: Response, next: NextFunction): Promise<void> => {
  try {
    const currentUserId = req.user!.id;
    const article = await ArticleRepository.favorite(req.params.slug, currentUserId);
    if (!article) {
      res.status(404).json({ errors: { body: ['Article not found'] } });
      return;
    }
    res.status(200).json({ article });
  } catch (error) {
    next(error);
  }
});

// 8. Unfavorite Article
router.delete('/articles/:slug/favorite', authenticate, async (req: AuthenticatedRequest, res: Response, next: NextFunction): Promise<void> => {
  try {
    const currentUserId = req.user!.id;
    const article = await ArticleRepository.unfavorite(req.params.slug, currentUserId);
    if (!article) {
      res.status(404).json({ errors: { body: ['Article not found'] } });
      return;
    }
    res.status(200).json({ article });
  } catch (error) {
    next(error);
  }
});

// 9. Add Comment
router.post('/articles/:slug/comments', authenticate, validate(NewCommentSchema), async (req: AuthenticatedRequest, res: Response, next: NextFunction): Promise<void> => {
  try {
    const currentUserId = req.user!.id;
    const comment = await CommentRepository.create(req.body.comment.body, req.params.slug, currentUserId);
    if (!comment) {
      res.status(404).json({ errors: { body: ['Article not found'] } });
      return;
    }
    res.status(201).json({ comment });
  } catch (error) {
    next(error);
  }
});

// 10. Get Comments
router.get('/articles/:slug/comments', optionalAuthenticate, async (req: AuthenticatedRequest, res: Response, next: NextFunction): Promise<void> => {
  try {
    const currentUserId = req.user?.id;
    const comments = await CommentRepository.getComments(req.params.slug, currentUserId);
    if (!comments) {
      res.status(404).json({ errors: { body: ['Article not found'] } });
      return;
    }
    res.status(200).json({ comments });
  } catch (error) {
    next(error);
  }
});

// 11. Delete Comment
router.delete('/articles/:slug/comments/:id', authenticate, async (req: AuthenticatedRequest, res: Response, next: NextFunction): Promise<void> => {
  try {
    const currentUserId = req.user!.id;
    const commentId = parseInt(req.params.id);
    const success = await CommentRepository.delete(commentId, currentUserId);
    if (!success) {
      res.status(404).json({ errors: { body: ['Comment not found'] } });
      return;
    }
    res.status(204).end();
  } catch (error: any) {
    if (error.message === 'Forbidden') {
      res.status(403).json({ errors: { body: ['You are not authorized to delete this comment'] } });
    } else {
      next(error);
    }
  }
});

export default router;
